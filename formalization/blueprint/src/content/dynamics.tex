\chapter{2-Adic Dynamics, Transfer Operators, and Dynamical Zeta Factorizations}
\label{chap:dynamics}

This chapter details the dynamical systems framework of the adèlic Collatz transfer operators acting on the 2-adic integers $\Ztwo = \varprojlim \Z/2^n\Z$. We formalize the transfer operators, the exact Fredholm determinant factorizations into cyclotomic towers, the spectral circle theorem, chiral eigenvalue pairings, ergodic averages, and the algebraic roadblocks in multi-step inductive towers.

\section{The 2-Adic Dynamical System and Transfer Operators}

\begin{definition}[2-Adic Collatz Map and Inverse Branches]
\label{def:collatz_z2}
\lean{Formalization.Dynamics.CollatzZ2.collatzTransferOp}
\leanok
The 2-adic Collatz map $T : \Ztwo \to \Ztwo$ is defined by:
\[
T(x) = \begin{cases}
x / 2 & \text{if } x \equiv 0 \pmod 2, \\
(3x + 1) / 2 & \text{if } x \equiv 1 \pmod 2.
\end{cases}
\]
The two continuous inverse branches $\psi_0, \psi_1 : \Ztwo \to \Ztwo$ are:
\[
\psi_0(y) = 2y, \quad \psi_1(y) = \frac{2y - 1}{3},
\]
which satisfy $T(\psi_0(y)) = y$ on the even residue set and $T(\psi_1(y)) = y$ on the odd residue set.
\end{definition}

\begin{definition}[Profinite Tower of Residue Rings]
\label{def:profinite_tower}
\lean{Formalization.Dynamics.ProfiniteTower.Z_2_Profinite}
\leanok
The projective system of finite rings $(\Z / 2^n \Z, \pi_{n, m})_{n \ge m \ge 1}$ with canonical quotient projections $\pi_{n, m} : \Z/2^n\Z \to \Z/2^m\Z$ has the projective limit:
\[
\Ztwo \cong \varprojlim_{n} \Z / 2^n \Z.
\]
\end{definition}

\begin{definition}[Directed Collatz Relation Matrix $D_n$]
\label{def:collatz_dir_matrix}
\lean{Formalization.Dynamics.CollatzRelMatrix.collatzDirMatrix}
\leanok
For each level $n \ge 1$, the directed relation matrix $D_n \in \Mat_{2^n \times 2^n}(\Z)$ is defined by:
\[
(D_n)_{x, y} = \begin{cases}
1 & \text{if } y = 2x \pmod{2^n} \text{ or } y = 2x + 1 \pmod{2^n}, \\
0 & \text{otherwise}.
\end{cases}
\]
Under the directed sheet involution $\tau_{\mathrm{dir}}(x) = x + 2^{n-1} \pmod{2^n}$, the matrix $D_n$ satisfies the fiber sum invariance $\sum_{y \in \pi^{-1}(z)} (D_n)_{x, y} = 2$.
\end{definition}

\section{Covering Factorizations and Block Diagonalization}

\begin{theorem}[Directed Covering Block Diagonalization]
\label{thm:collatz_dir_block_diag}
\uses{def:collatz_dir_matrix}
\lean{Formalization.Dynamics.CollatzRelMatrix.spectral_tower_one_step}
\leanok
Let $D_n'$ be the lifted 2-sheet directed transition matrix. Under the normalized Hadamard similarity transform $H$, $D_n'$ decomposes into block diagonal form:
\[
H D_n' H^{-1} = \begin{pmatrix} D_{\mathrm{weighted}} & 0 \\ 0 & D_{\mathrm{twisted}} \end{pmatrix},
\]
where $D_{\mathrm{weighted}} \in \Mat_{2^{n-1} \times 2^{n-1}}(\Z)$ is the projected base transition matrix and $D_{\mathrm{twisted}} \in \Mat_{2^{n-1} \times 2^{n-1}}(\Z)$ represents the twisted/antisymmetric sector.
\end{theorem}

\begin{theorem}[Covering Determinant Factorization]
\label{thm:covering_det_factorization}
\uses{thm:collatz_dir_block_diag}
\lean{Formalization.Dynamics.CoveringFactorization.det_collatzDirMatrix_factorization}
\leanok
The Fredholm determinant of the lifted covering factors multiplicatively:
\[
\det(I - t D_n') = \det(I - t D_{\mathrm{weighted}}) \cdot \det(I - t D_{\mathrm{twisted}}).
\]
\end{theorem}

\begin{theorem}[Chiral Anti-Commutation and Paired Spectra]
\label{thm:chiral_decomposition}
\uses{thm:collatz_dir_block_diag}
\lean{Formalization.Dynamics.ChiralDecomposition.chiral_anticommute}
\leanok
There exists an involutive chiral grading operator $\Gamma$ ($\Gamma^2 = I$) on the antisymmetric subspace such that the sheet difference operator $D_{\mathrm{diff}}$ satisfies the exact anti-commutation relation:
\[
\{ \Gamma, D_{\mathrm{diff}} \} = \Gamma D_{\mathrm{diff}} + D_{\mathrm{diff}} \Gamma = 0.
\]
Consequently, all non-zero eigenvalues of $D_{\mathrm{diff}}$ occur in exact chiral pairs $\{\lambda, -\lambda\}$.
\end{theorem}

\section{Cyclic Weight Matrices and Cyclotomic Towers}

\begin{theorem}[Cyclic Weight Characteristic Polynomial]
\label{thm:cyclic_weight_charpoly}
\lean{Formalization.Dynamics.CyclicWeightCharpoly.charpoly_cyclicWeightMatrix}
\leanok
Let $C(w) \in \Mat_{m \times m}(R)$ be a cyclic bidiagonal weight matrix with weights $w = (w_0, w_1, \ldots, w_{m-1})$. The characteristic polynomial of $C(w)$ is:
\[
\det(X I - C(w)) = X^m - \prod_{j=0}^{m-1} w_j.
\]
\end{theorem}

\begin{theorem}[Cyclotomic Product of Primitive Roots]
\label{thm:cyclotomic_product}
\uses{thm:cyclic_weight_charpoly}
\lean{Formalization.Dynamics.CyclotomicProduct.W_1_mul_W_2_eq_two}
\leanok
For any power of two $2^n$ ($n \ge 2$), let $\mu_{2^n}^*$ denote the set of primitive $2^n$-th roots of unity. Then:
\[
\prod_{\zeta \in \mu_{2^n}^*} (1 - \zeta) = 2, \quad \text{and} \quad \prod_{\zeta \in \mu_{2^n}^*} |1 - \zeta|^2 = 4.
\]
\end{theorem}

\begin{theorem}[Dynamical Zeta Function Fredholm Factorization]
\label{thm:dynamical_zeta_factorization}
\uses{thm:cyclic_weight_charpoly, thm:cyclotomic_product, thm:covering_det_factorization}
\lean{Formalization.Dynamics.DynamicalZetaFactorization.fredholmFactorization}
\leanok
The Fredholm determinant of the Collatz transfer operator at level $n$ factors completely into cyclotomic polynomials:
\[
\det(I - t D_n) = (1 - t)(1 - 2t) \prod_{k=2}^n \Phi_{2^k}(t),
\]
where $\Phi_{2^k}(t) = 1 + t^{2^{k-1}}$ is the $2^k$-th cyclotomic polynomial.
The total number of non-zero spectral modes is $\sum_{k=1}^{n-1} 2^{k-1} = 2^{n-1} - 1$, exactly matching the rank of the nilpotent Jordan blocks.
\end{theorem}

\begin{theorem}[Rationality of the Collatz and Automaton Zeta Functions]
\label{thm:automaton_zeta_rationality}
\uses{thm:dynamical_zeta_factorization}
\lean{Formalization.Dynamics.AutomatonZeta.collatz_zeta_rationality}
\leanok
The dynamical zeta function $Z_n(t) = \exp\left(\sum_{m=1}^\infty \frac{t^m}{m} \Tr(D_n^m)\right)$ is a rational function in $t$:
\[
Z_n(t) = \frac{1}{\det(I - t D_n)} \in \Q(t).
\]
\end{theorem}

\section{The Spectral Circle Theorem}

\begin{theorem}[The Spectral Circle Theorem]
\label{thm:spectral_circle}
\uses{thm:cyclotomic_product, thm:dynamical_zeta_factorization}
\lean{Formalization.Dynamics.SpectralCircle.spectral_circle}
\leanok
Let $\lambda \in \C$ be any non-zero eigenvalue of the twisted sector transfer operator $D_{\mathrm{twisted}}$. Then:
\[
|\lambda| = \sqrt{2},
\]
and the complex arguments of the eigenvalues are uniformly distributed at the angles $\theta_j = \frac{2\pi j}{2^n}$ corresponding to the roots of the cyclotomic tower.
\end{theorem}

\section{Ergodic Averages, Mixing, and Shadowing}

\begin{theorem}[$L^2$ Mixing Decay of Transfer Operators]
\label{thm:l2_mixing}
\uses{thm:collatz_dir_block_diag}
\lean{Formalization.Dynamics.L2Mixing.L2_decay_bound}
\leanok
Let $P_n$ be the normalized Markov transition operator on $L^2(\Z/2^n\Z)$. For any mean-zero function $f \in L^2_0$, the $k$-step transition satisfies the exponential mixing decay:
\[
\| P_n^k f \|_{L^2} \le C \rho^k \| f \|_{L^2},
\]
where $\rho = 1 / \sqrt{2} < 1$.
\end{theorem}

\begin{theorem}[Mean Ergodic Theorem on 2-Adic Integers]
\label{thm:mean_ergodic}
\uses{def:collatz_z2, thm:l2_mixing}
\lean{Formalization.Dynamics.MeanErgodic.A_N_converges_L2}
Let $A_N = \frac{1}{N} \sum_{k=0}^{N-1} U^k$ denote the Ces\`aro averages of the Koopman isometry $U f = f \circ T$ on $L^2(\Ztwo, \mu_2)$. Then $A_N$ converges strongly in $L^2$ and in measure to the orthogonal projection onto the invariant subspace of constant functions:
\[
\lim_{N \to \infty} A_N f = \left(\int_{\Ztwo} f \, d\mu_2\right) \mathbf{1}.
\]
\end{theorem}

\begin{theorem}[Orbit Shadowing on 2-Adic State Spaces]
\label{thm:orbit_shadowing}
\uses{def:collatz_z2}
\lean{Formalization.Dynamics.OrbitShadowing.deterministic_avg}
\leanok
Every $\delta$-pseudo-orbit of the Collatz transfer walk on $\Z/2^n\Z$ is shadowed by an exact 2-adic deterministic trajectory with average tracking error bounded by $O(2^{-n})$.
\end{theorem}

\section{Conjectures and Inductive Obstructions}

\begin{theorem}[Inductive Multi-Step Obstruction: Double Edges]
\label{thm:inductive_tower_obstruction}
\uses{def:schreier_graph_gd}
\lean{Formalization.Dynamics.InductiveTower.spectral_tower_two_step_fails}
\leanok
In the 2-step inductive tower $\Z/2^d\Z \to \Z/2^{d-2}\Z$, the quotient multigraph develops multiple parallel edges ($A_{\mathrm{weighted}} \neq A$). Consequently, naive inductive recursion fails beyond step 1 without dynamic characteristic polynomial tracking over rings.
\end{theorem}

\begin{conjecture}[Twisted Block Power Conjecture]
\label{conj:twisted_block_pow}

\[
(D_{\mathrm{twisted}})^{2^{n-1}} = I_{2^{n-1}} \pmod{2^n}.
\]
\end{conjecture}
